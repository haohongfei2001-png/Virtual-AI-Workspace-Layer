/** Actual binding service and ingestion/UI owners; synthetic browser/provider and MockIDB. */
import test from 'node:test';
import assert from 'node:assert/strict';
import {createHostBindingService} from '../../src/runtime/host-binding-service.mjs';
import {createURLPolicy} from '../../src/host/url-policy.mjs';
import {ScopeRepository} from '../../src/b1/scope-repository.mjs';
import {MockIDBFactory} from '../b23/mock-idb.mjs';
const drain=async()=>{for(let i=0;i<80;i++)await Promise.resolve();};
const deferred=()=>{let resolve,reject;const promise=new Promise((a,b)=>{resolve=a;reject=b;});return {promise,resolve,reject};};
function clock(){let time=0,id=0,reads=0;const timers=new Map();return {
 now:()=>{reads++;return 1000+time;},elapsedNow:()=>{reads++;return time;},
 schedule(fn,ms){const key=++id;timers.set(key,{fn,at:time+ms});return key;},cancel:key=>timers.delete(key),
 get reads(){return reads;},get time(){return time;},timers,
 async advance(ms){time+=ms;for(const [key,timer]of [...timers])if(timer.at<=time&&timers.delete(key))timer.fn();await drain();}
};}
async function fixture(t){
 const timer=clock(),repository=new ScopeRepository({indexedDB:new MockIDBFactory(),databaseName:'pass-expiry'});let id=0,seq=0;
 const sender={id:'ext',tab:{id:1,incognito:false},frameId:0,documentId:'doc',documentLifecycle:'active',url:'https://chatgpt.com/'};
 const frame={documentId:'doc',url:sender.url,documentLifecycle:'active',parentFrameId:-1};let nextFrame=null;
 const profile={id:'offline',revision:1,urlPolicy:createURLPolicy()};
 const service=createHostBindingService({...timer,extensionId:'ext',repository,uiURL:'chrome-extension://ext/ui.html',surfaceProfiles:new Map([[profile.id,profile]]),
  getFrame:async()=>{if(nextFrame){const callback=nextFrame;nextFrame=null;await callback();}return frame;},allowedURL:u=>u===sender.url,
  provider:{read:async()=>({principalId:'synthetic',expiresAt:new Date(1000000).toISOString(),contexts:[{id:'personal',kind:'personal'}],providerProfile:'test',transport:'worker-fixed-fetch'})},profile:{id:'test',singlePersonalDefault:true},makeId:()=>`id_${++id}`});
 t.after(()=>{service.stop();repository.close();});
 const hello=await service.handle({v:1,type:'HOST_HELLO',clientId:'client',seq:++seq},sender);assert.equal(hello.ok,true);
 const {documentToken,producerToken,routeEpoch}=hello.binding,auth={documentToken,producerToken,routeEpoch};
 const refresh=()=>service.handle({v:1,type:'HOST_REFRESH',clientId:'client',seq:++seq,...auth},sender);
 assert.equal((await refresh()).state,'verified');assert.equal((await service.handle({v:1,type:'HOST_UI_MOUNT',...auth,mountId:'mount',regionEpoch:1},sender)).ok,true);
 return {service,timer,sender,auth,frame,holdNextFrame:fn=>{nextFrame=fn;},
  send:(phase='start',passSeq=1)=>service.handle({v:1,type:'HOST_INGESTION_PASS',...auth,profileId:'offline',profileVersion:1,passSeq,phase},sender),
  idle:()=>service.handle({v:1,type:'HOST_UI_IDLE_CHECK',...auth,mountId:'mount'},sender),
  async renew(){assert.equal((await service.handle({v:1,type:'HOST_DOCUMENT_RENEW',clientId:'client',seq:++seq,...auth},sender)).ok,true);assert.equal((await refresh()).state,'verified');}
 };
}
function pureIdle(f,expected){const before=f.timer.reads;for(let i=0;i<10;i++)assert.equal(f.service.isIdle(),expected);assert.equal(f.timer.reads,before,'idle query does not sample clocks');}

test('accepted start without terminal ends only its scheduling hint at original deadline',async t=>{
 const f=await fixture(t);assert.equal((await f.send()).ok,true);pureIdle(f,false);
 await f.timer.advance(15000);await f.renew();await f.timer.advance(14999);pureIdle(f,false);assert.deepEqual(await f.idle(),{ok:false,reason:'ui-release-busy'});
 await f.timer.advance(1);pureIdle(f,true);assert.deepEqual(await f.idle(),{ok:true});
 assert.equal((await f.send()).ok,false,'expiry tombstone rejects replay; no synthetic end evidence');
});

test('duplicate accepted start and document renewal cannot extend pass deadline',async t=>{
 const f=await fixture(t);await f.send();await f.timer.advance(15000);await f.renew();await f.timer.advance(14999);assert.equal((await f.send()).ok,true);pureIdle(f,false);
 await f.timer.advance(1);pureIdle(f,true);assert.equal((await f.send()).ok,false);
});

for(const phase of ['end','cancel'])test(phase+' retains tombstone and clears timer without clock reads in idle',async t=>{
 const f=await fixture(t);await f.send();assert.equal((await f.send(phase)).ok,true);pureIdle(f,true);assert.equal(f.timer.timers.size,0);assert.equal((await f.send()).ok,false);
});

test('slow start validation uses entry deadline, not a fresh thirty seconds at acceptance',async t=>{
 const f=await fixture(t),gate=deferred();let entered=false;f.holdNextFrame(()=>{entered=true;return gate.promise;});
 const start=f.send();await drain();assert.equal(entered,true);await f.timer.advance(5000);gate.resolve();assert.equal((await start).ok,true);
 await f.timer.advance(10000);await f.renew();await f.timer.advance(15000);pureIdle(f,true);
});


test('old pass timer cannot expire its newer replacement or erase replay tombstones',async t=>{
 const f=await fixture(t);await f.send();const old=[...f.timer.timers.values()].find(x=>x.at===30000);assert(old,'original expiry timer');
 await f.timer.advance(15000);await f.renew();assert.equal((await f.send('start',2)).ok,true);old.fn();pureIdle(f,false);
 assert.equal(f.timer.timers.size,1,'one current bounded timer');await f.timer.advance(15000);pureIdle(f,false);await f.renew();
 assert.equal((await f.send('end',1)).ok,false);pureIdle(f,false);await f.timer.advance(15000);pureIdle(f,true);assert.equal((await f.send('start',2)).ok,false);
});

for(const outcome of ['resolve','reject'])test('expiry preserves underlying start validation after RPC timeout until '+outcome,async t=>{
 const f=await fixture(t);await f.send();await f.timer.advance(15000);await f.renew();await f.timer.advance(14000);
 const gate=deferred();let entered=false;f.holdNextFrame(()=>{entered=true;return gate.promise;});const later=f.send('start',2);await drain();assert(entered);
 await f.timer.advance(1000);pureIdle(f,false,'first hint expired but second browser validation is still real work');
 await f.timer.advance(5000);assert.equal((await later).ok,false);pureIdle(f,false,'timed-out raw browser work remains');
 if(outcome==='resolve')gate.resolve();else gate.reject(Error('late-browser-rejection'));await drain();pureIdle(f,true);
});

for(const action of ['invalidateDocument','invalidateTab','tabClosed','stop'])test(action+' cancels the owned hint timer and stale callback cannot revive state',async t=>{
 const f=await fixture(t);await f.send();const old=[...f.timer.timers.values()].find(x=>x.at===30000);assert(old,'original expiry timer');
 if(action==='invalidateDocument')f.service.invalidate({tabId:1,documentId:'doc'});
 else if(action==='invalidateTab')f.service.invalidate({tabId:1});else if(action==='tabClosed')f.service.tabClosed(1);else f.service.stop();
 assert.equal(f.timer.timers.size,0);old.fn();pureIdle(f,action!=='stop');assert.equal(f.timer.timers.size,0);assert.equal((await f.send()).ok,false);
});
