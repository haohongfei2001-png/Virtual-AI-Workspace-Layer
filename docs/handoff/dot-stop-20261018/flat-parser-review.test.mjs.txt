const C='11111111-1111-4111-8111-111111111111',now=Date.parse('2026-10-18T00:00:00Z');
const principal='review-principal',contextId='review-personal';

const profile={id:'review-flat-profile',conversationRequestContext:'conversation-detail-plural-v1',conversationMetadataFormat:'message-list-v1',currentConversationReadEnabled:true,currentConversationContextHeader:true};
const row=i=>({id:'m'+i,author:{role:'user',id:principal},create_time:1700000000,content:{parts:['SYNTHETIC_BODY_MUST_NOT_ESCAPE']}});
const sequence=()=>Array.from({length:308},(_,i)=>row(i));
function fixture(messages,wanted,{holdDetail=false}={}){
 let current=true,release,entered;const atDetail=new Promise(r=>entered=r),held=new Promise(r=>release=r),abort=new AbortController(),calls=[],results=[];
 const provider=createSessionProvider({profile,now:()=>now,elapsedNow:()=>0,fetch:async(url,options)=>{
  calls.push({url,options});const n=calls.length;
  if(n===3){entered();if(holdDetail)await held;}
  const body=n===1?{user:{id:principal},accessToken:'SYNTHETIC_TOKEN',expires:'2026-10-18T01:00:00Z'}:n===2?{accounts:{[contextId]:{account:{account_id:contextId,structure:'personal'}}}}:{conversation_id:C,messages,current_node:'m307',page_info:{has_next_page:false}};
  const response=new Response(JSON.stringify(body),{headers:{'content-type':'application/json'}});Object.defineProperty(response,'url',{value:url});return response;
 }});
 const original=provider.readCurrentConversation;
 const wrapped={...provider,readCurrentConversation:async args=>{const result=await original(args);results.push(result);return result;}};
 const context={scope:{accountId:principalKey,organizationId:contextKey},conversationId:C,messageIds:wanted,bodyEpoch:1,providerProfile:profile.id,providerTransport:'worker-fixed-fetch',signal:abort.signal,guard:()=>current};
 const observations={async withCurrent(proof,work){assert.equal(proof,'proof');if(!current)throw Error('stale-body-proof');const value=await work(context);if(!current)throw Error('stale-body-proof');return value;}};
 const reader=createCurrentMetadataReader({observations,provider:wrapped,enabled:true,now:()=>now});
 return {reader,provider,calls,results,atDetail,release,revoke(){current=false;abort.abort();}};
}
const ids=value=>value.metadata.messages.map(x=>x.messageId);
function invariant(value,f){
 assert.equal(value.state,'partial');assert.equal(value.metadata.coverage,'observed-only');assert(value.metadata.gaps.some(g=>g.reason==='lineage-unavailable'));
 assert(value.metadata.messages.every(m=>m.mappingNodeId===null&&m.lineage===null));
 assert(!/SYNTHETIC_BODY_MUST_NOT_ESCAPE|SYNTHETIC_TOKEN|sentAt|ackWitness|actorVerified/.test(JSON.stringify(value)));
 assert.equal(f.calls.length,3);assert(f.calls[2].url.endsWith('/backend-api/conversations/'+C));
 assert.equal(f.provider.acceptanceFor(f.results[0]),null);assert.equal(f.provider.historyFor(f.results[0]),null);
}
test('308 rows: current IDs after prefix cap remain missing, never falsely associated',async()=>{
 const wanted=['m303','m304','m305','m306','m307'],f=fixture(sequence(),wanted);
 try{const value=await f.reader.read('proof');invariant(value,f);assert.deepEqual(ids(value),[]);assert(value.metadata.gaps.some(g=>g.reason==='message-limit'));assert(value.metadata.gaps.some(g=>g.reason==='current-message-missing'));}finally{f.reader.stop();}
});
test('same 308 identities with current IDs first yield exactly those IDs, still partial',async()=>{
 const rows=sequence(),wanted=['m303','m304','m305','m306','m307'],f=fixture([...rows.slice(303),...rows.slice(0,303)],wanted);
 try{const value=await f.reader.read('proof');invariant(value,f);assert.deepEqual(ids(value),wanted);assert(!value.metadata.gaps.some(g=>g.reason==='current-message-missing'));await assert.rejects(f.reader.withReadback(value,'proof',wanted[0],()=>assert.fail('flat ACK')),/unavailable-send-readback/);}finally{f.reader.stop();}
});
